Console.Write("Введите сумму вклада: ");
decimal sum = Convert.ToDecimal(Console.ReadLine());

Console.Write("Введите количество месяцев: ");   //decimal sum = — объявляет переменную sum с типом decimal,
int months = Convert.ToInt32(Console.ReadLine());   //Convert.ToDecimal() — статический метод класса Convert, преобразует строку в тип decimal
                                                    //Convert.ToInt32 - преобразует значение одного типа данных в 32-битное целое число
for (int i = 0; i < months; i++)   
{
    sum += sum * 0.07m;   // цикл for в C#, который используется для расчёта будущей стоимости инвестиции с учётом сложных процентов
}

Console.WriteLine($"Конечная сумма вклада: {sum}");
